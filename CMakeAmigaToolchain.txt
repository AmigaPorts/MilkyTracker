list(APPEND CMAKE_MODULE_PATH ${PROJECT_SOURCE_DIR}/cmake)

set(CMAKE_SYSTEM_NAME AmigaOS3)
set(CMAKE_SYSTEM_PROCESSOR m68k)

set(CMAKE_SYSROOT /opt/amiga)

set(tools ${CMAKE_SYSROOT})

set(CMAKE_FIND_ROOT_PATH_MODE_PROGRAM NEVER)
set(CMAKE_FIND_ROOT_PATH_MODE_LIBRARY ONLY)
set(CMAKE_FIND_ROOT_PATH_MODE_INCLUDE ONLY)
set(CMAKE_FIND_ROOT_PATH_MODE_PACKAGE ONLY)

set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)

set(AMIGA 1)
set(AMIGAOS3 1)

if(DEFINED AMIGA_CPU)
    set(ARCH ${AMIGA_CPU})
else()
    set(ARCH 68040)
endif()

add_compile_definitions(AMIGA_ARCH=${ARCH})

message(STATUS "[m68k-amigaos] Compiling for ${ARCH} @ ${NOW}")

set(CMAKE_FIND_LIBRARY_SUFFIXES ".a")
set(CMAKE_C_COMPILER ${tools}/bin/m68k-amigaos-gcc)
set(CMAKE_CXX_COMPILER ${tools}/bin/m68k-amigaos-g++)
set(CMAKE_AR ${tools}/bin/m68k-amigaos-ar)

# Compiler flags
set(CMAKE_C_FLAGS_DEBUG "${CMAKE_C_FLAGS_DEBUG} -m${ARCH} -mhard-float -Os -fomit-frame-pointer -fno-exceptions -s -noixemul")
set(CMAKE_C_FLAGS_RELEASE "${CMAKE_C_FLAGS_RELEASE} -m${ARCH} -mhard-float -Os -fomit-frame-pointer -fno-exceptions -s -noixemul")
set(CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG} -m${ARCH} -mhard-float -Os -fomit-frame-pointer -fno-exceptions -fpermissive -fno-rtti -s -noixemul")
set(CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE} -m${ARCH} -mhard-float -Os -fomit-frame-pointer -fno-exceptions -fpermissive -fno-rtti -s -noixemul")
set(BUILD_SHARED_LIBS OFF)

# Linker configuration
set(CMAKE_EXE_LINKER_FLAGS "-noixemul -ldebug -s -Xlinker --allow-multiple-definition ")
set(CMAKE_SHARED_LIBRARY_LINK_C_FLAGS "")
set(CMAKE_SHARED_LIBRARY_LINK_CXX_FLAGS "")
